cmake_minimum_required(VERSION 3.10)
project(cabinet_npu_v4 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "Use the board ARM64 Linux SDK/toolchain; a host build is not a board adapter")
endif()
foreach(required SDK_INCLUDE ARMNN_LIBRARY ONNX_PARSER_LIBRARY)
    if(NOT DEFINED ${required} OR NOT EXISTS "${${required}}")
        message(FATAL_ERROR "Provide the existing vendor SDK path: ${required}")
    endif()
endforeach()
foreach(library ARMNN_LIBRARY ONNX_PARSER_LIBRARY)
    if(IS_DIRECTORY "${${library}}")
        message(FATAL_ERROR "Provide an actual vendor library file: ${library}")
    endif()
endforeach()
add_library(edge_npu_logits_ps SHARED edge_npu_logits_ps.cpp)
set_target_properties(edge_npu_logits_ps PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
target_compile_options(edge_npu_logits_ps PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
target_include_directories(edge_npu_logits_ps PRIVATE "${SDK_INCLUDE}")
target_link_libraries(edge_npu_logits_ps PRIVATE "${ARMNN_LIBRARY}" "${ONNX_PARSER_LIBRARY}")
get_filename_component(SDK_LIBRARY_DIR "${ARMNN_LIBRARY}" DIRECTORY)
set_property(TARGET edge_npu_logits_ps APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--no-undefined -Wl,-rpath-link,${SDK_LIBRARY_DIR}")
option(BUILD_LEGACY_ABI2 "Build the legacy full-graph ABI2 adapter, not the R30 release" OFF)
if(BUILD_LEGACY_ABI2)
    add_library(edge_npu_v4 SHARED edge_npu_v4.cpp)
    set_target_properties(edge_npu_v4 PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
    target_compile_options(edge_npu_v4 PRIVATE -Wall -Wextra -Werror)
    target_include_directories(edge_npu_v4 PRIVATE "${SDK_INCLUDE}")
    target_link_libraries(edge_npu_v4 PRIVATE "${ARMNN_LIBRARY}" "${ONNX_PARSER_LIBRARY}")
endif()
